cmake_minimum_required(VERSION 3.16)
project(canrpc_core LANGUAGES C)
add_library(canrpc_core INTERFACE)
target_include_directories(canrpc_core INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(canrpc_core INTERFACE c_std_99)
option(CANRPC_CORE_TESTS "Build portable protocol tests" ON)
if(CANRPC_CORE_TESTS)
  enable_testing()
  add_executable(test_canrpc_core test/test_core.c)
  target_link_libraries(test_canrpc_core PRIVATE canrpc_core)
  target_compile_options(test_canrpc_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME canrpc_core COMMAND test_canrpc_core)
  add_executable(test_canrpc_rpc test/test_rpc.c)
  target_link_libraries(test_canrpc_rpc PRIVATE canrpc_core)
  target_compile_options(test_canrpc_rpc PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME canrpc_rpc COMMAND test_canrpc_rpc)
endif()
install(FILES include/canrpc_core.h DESTINATION include)
