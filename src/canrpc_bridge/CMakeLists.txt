cmake_minimum_required(VERSION 3.8)
project(canrpc_bridge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(std_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/Call.srv" "msg/BridgeStatus.msg")
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
add_library(canrpc_config STATIC src/config.cpp)
target_compile_features(canrpc_config PRIVATE cxx_std_17)
target_include_directories(canrpc_config PUBLIC include)
target_link_libraries(canrpc_config yaml-cpp)
add_executable(canrpc_bridge_node src/canrpc_bridge.cpp src/socketcan.cpp)
target_compile_features(canrpc_bridge_node PRIVATE cxx_std_17)
target_compile_options(canrpc_bridge_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(canrpc_bridge_node PRIVATE include core/include)
set_target_properties(canrpc_bridge_node PROPERTIES OUTPUT_NAME canrpc_bridge
  RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/bin)
target_link_libraries(canrpc_bridge_node canrpc_config "${cpp_typesupport_target}")
ament_target_dependencies(canrpc_bridge_node rclcpp rclcpp_lifecycle std_msgs lifecycle_msgs)
install(TARGETS canrpc_bridge_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ core/include/ DESTINATION include PATTERN "__pycache__" EXCLUDE)
install(DIRECTORY launch config tools docs core adapters DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_test REQUIRED)
  add_executable(test_protocol test/test_protocol.c)
  target_compile_features(test_protocol PRIVATE c_std_99)
  target_compile_options(test_protocol PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(test_protocol PRIVATE include core/include)
  add_test(NAME canrpc_protocol COMMAND test_protocol)
  foreach(core_test core rpc)
    add_executable(test_portable_${core_test} core/test/test_${core_test}.c)
    target_compile_features(test_portable_${core_test} PRIVATE c_std_99)
    target_compile_options(test_portable_${core_test} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_include_directories(test_portable_${core_test} PRIVATE core/include)
    add_test(NAME canrpc_portable_${core_test} COMMAND test_portable_${core_test})
  endforeach()
  add_executable(test_config test/test_config.cpp)
  target_compile_features(test_config PRIVATE cxx_std_17)
  target_link_libraries(test_config canrpc_config)
  add_test(NAME canrpc_config COMMAND test_config
    ${CMAKE_CURRENT_SOURCE_DIR}/config/gakusai.yaml
    ${CMAKE_CURRENT_SOURCE_DIR}/config/reuse_example.yaml)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
